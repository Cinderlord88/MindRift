  // Read only EXPLICIT personal-fear declarations from character Story Cards.
  // Supported lines: Fear: ..., Flaw: ..., Personal Fear: ..., Weakness: ...
  // This is a player-authored override, NOT AI interpretation of prose.
  cardFear(card) {
    const fields = [card.entry, card.description, card.text]
      .filter(value => typeof value === 'string' && value.trim());
    const content = fields.join('\n');
    const match = content.match(/(?:^|\n)\s*(?:(?:personal|psychological)\s+)?(?:fear|flaw|weakness)\s*:\s*([^\r\n]+)/i);
    return match ? match[1].trim().slice(0, 180) : '';
  },

  // 'Fear of losing control over her own body'). Lasts across future turns.
  setPersonalFear(name, description) {
    this.ensure();
    const person = this.find(name);
    const value = String(description || '').trim();
    if (!person || !value) return false;
    person.personalFearOverride = value.slice(0, 180);
    person.weakness = person.personalFearOverride;
    person.personalFearSource = 'manual';
    person.personalFearAssignedValue = person.weakness;
    return true;
  },

      const pool = this.relaxationActivities.slice();
      const selected = [];
      for (let i = 0; i < 3 && pool.length; i++) {

      const pool = this.selfHealingActivities.slice();

  // ================================================================
  // 1. PERSISTENT DATABASE
  // ================================================================
  // Script state is the source of truth; developed tendencies may sync to Story Cards.
  // Existing scores survive subsequent turns and card scans.
  init() {
    if (!state.psyEngine) {
      state.psyEngine = {
        people: {},       // Character records, indexed by card ID or name
        pending: [],      // Psychological effects awaiting narration
        seen: {},         // Event markers processed during the current turn
        turn: 0
      };
    }

    return state.psyEngine;
  },

  // ================================================================
  // 2. FIND CHARACTER STORY CARDS
  // ================================================================
  cards() {
    if (typeof storyCards === 'undefined' || !Array.isArray(storyCards)) {
      return [];
    }

    return storyCards.filter(card => {
      const cardType = String(card.type || '');
      return /^(character|characters)$/i.test(cardType);
    });
  },

  // Prefer a card title; otherwise use the first comma-separated key.
  name(card) {
    const firstKey = String(card.keys || '').split(',')[0];
    return String(card.title || firstKey).trim();
  },

  // The name stored in the character's derangements array is mechanical.
  // Migrate older entries such as "Kleptomania - Compulsion to steal" safely.
  derangementName(entry) {
    const raw = typeof entry === 'string' ? entry :
      (entry && typeof entry.name === 'string' ? entry.name : '');
    return String(raw).split(/\s+-\s+/)[0].trim();
  },

  derangementProfile(entry) {
    const name = this.derangementName(entry).toLowerCase();
    return this.derangementProfiles.find(item => item.name.toLowerCase() === name) || null;
  },

  // Return JUST the personality language, never clinical labels or commands.
  developedTendencies(character) {
    const current = Array.isArray(character && character.derangements)
      ? character.derangements : [];
    const seen = new Set();
    const text = [];
    for (const item of current.slice(0, 5)) {
      const name = this.derangementName(item).toLowerCase();
      if (!name || seen.has(name)) continue;
      seen.add(name);
      const profile = this.derangementProfile(item);
      if (profile) text.push(profile.storyCardText);
      else psyLog('PSY DERANGEMENT TEXT UNAVAILABLE: ' +
        String(character && character.name || 'Unknown') + ' | ' + name);
    }
    return text;
  },

  // ================================================================
  // 2a. SEPARATE PSYCHOLOGY STORY CARDS
  // ================================================================
  // This engine never writes to original Character / Characters Story Cards.
  // A single PSYCHOLOGY card represents all developed tendencies for a person.
  // The unique first trigger doubles as the card's requested name.
  psychologyProfileTitle(character) {
    return String(character && character.name || '').trim() + ' - Psych Profile';
  },

  // Triggers have a finite budget. One unique first key plus at most four
  // inherited keys (not all original keys), with a total length cap.
  psychologyProfileKeys(character, sourceCard) {
    const MAX_KEYS = 5;
    const MAX_TOTAL_LENGTH = 180;
    const nameKey = this.psychologyProfileTitle(character);
    if (!nameKey || nameKey.length > MAX_TOTAL_LENGTH) return '';
    const chosen = [nameKey];
    const seen = new Set([nameKey.toLowerCase()]);
    const originals = String(sourceCard && sourceCard.keys || '').split(',');
    for (const rawKey of originals) {
      if (chosen.length >= MAX_KEYS) break;
      const key = rawKey.replace(/[\r\n]+/g, ' ').trim();
      if (!key || seen.has(key.toLowerCase())) continue;
      if (chosen.join(', ').length + 2 + key.length > MAX_TOTAL_LENGTH) continue;
      chosen.push(key);
      seen.add(key.toLowerCase());
    }
    return chosen.join(', ');
  },

  psychologyProfileEntry(character, sourceOfDerangements) {
    const tendencies = this.developedTendencies(sourceOfDerangements || character);
    if (!tendencies.length) return '';
    return 'Psychological profile for ' + character.name + '.\n' +
      'Developed tendencies: ' + tendencies.join(' ') + '\n' +
      'These tendencies can meaningfully influence behavior when relevant, ' +
      'but they do not override the character’s established personality or agency.';
  },

  // A persistent bookkeeping record, not a replacement for checking real cards.
  psychologyCardRecord(character) {
    if (!character.psychCard || typeof character.psychCard !== 'object' ||
        Array.isArray(character.psychCard)) {
      character.psychCard = {
        exists: false, name: '', triggers: '', entry: '', cardId: null
      };
    }
    const p = character.psychCard;
    if (typeof p.exists !== 'boolean') p.exists = false;
    if (typeof p.name !== 'string') p.name = '';
    if (typeof p.triggers !== 'string') p.triggers = '';
    if (typeof p.entry !== 'string') p.entry = '';
    if (p.cardId === undefined) p.cardId = null;
    return p;
  },

  syncPsychologyProfileCard(character, sourceOfDerangements) {
    if (!character || !character.name || typeof storyCards === 'undefined' ||
        !Array.isArray(storyCards)) return false;
    const record = this.psychologyCardRecord(character);
    const source = this.cards().find(card =>
      this.name(card).toLowerCase() === character.name.toLowerCase());
    if (!source) {
      psyLog('PSY PROFILE SKIPPED: Original character card missing for ' + character.name);
      return false;
    }

    const title = this.psychologyProfileTitle(character);
    const keys = this.psychologyProfileKeys(character, source);
    const entry = this.psychologyProfileEntry(character, sourceOfDerangements);
    if (!keys) {
      psyLog('PSY PROFILE SKIPPED: No room for unique trigger for ' + character.name);
      return false;
    }
    // Desired state is recorded even if AI Dungeon refuses the card operation.
    record.name = title;
    record.triggers = keys;
    record.entry = entry;

    // The card's type plus unique name (or known ID) defines ownership.
    // NEVER locate cards by ordinary character triggers alone.
    const matches = [];
    for (let i = 0; i < storyCards.length; i++) {
      const card = storyCards[i];
      if (String(card.type || '').toUpperCase() !== 'PSYCHOLOGY') continue;
      const first = String(card.keys || '').split(',')[0].trim();
      if (first.toLowerCase() === title.toLowerCase() ||
          String(card.title || '').trim().toLowerCase() === title.toLowerCase() ||
          (record.cardId != null && card.id != null &&
           String(card.id) === String(record.cardId))) {
        matches.push(i);
      }
    }

    // A prior interrupted run may have made duplicates. Delete only clearly
    // engine-authored extra cards; never silently delete someone's custom text.
    if (matches.length > 1 && typeof removeStoryCard === 'function') {
      for (const i of matches.slice(1).reverse()) {
        const old = storyCards[i];
        if (!String(old.entry || '').startsWith(
          'Psychological profile for ' + character.name + '.\n')) {
          psyLog('PSY PROFILE DUPLICATE: Non-engine card left untouched for ' + title);
          continue;
        }
        try { removeStoryCard(i); }
        catch (error) {
          psyLog('PSY PROFILE DUPLICATE CLEANUP FAILED: ' + title + ' | ' + String(error));
        }
      }
    }

    let index = storyCards.findIndex(card =>
      String(card.type || '').toUpperCase() === 'PSYCHOLOGY' &&
      (String(card.keys || '').split(',')[0].trim().toLowerCase() === title.toLowerCase() ||
       String(card.title || '').trim().toLowerCase() === title.toLowerCase() ||
       (record.cardId != null && card.id != null &&
        String(card.id) === String(record.cardId))));
    const found = index >= 0 ? storyCards[index] : null;
    record.exists = !!found;
    record.cardId = found && found.id != null ? found.id : null;

    if (!entry) {
      // No derangements: no Psychology card is necessary.
      if (!found) return true;
      if (!String(found.entry || '').startsWith(
          'Psychological profile for ' + character.name + '.\n')) {
        psyLog('PSY PROFILE REMOVE SKIPPED: Card contains unrecognized content for ' + title);
        return false;
      }
      if (typeof removeStoryCard !== 'function') return false;
      try {
        removeStoryCard(index);
        record.exists = false;
        record.cardId = null;
        psyLog('PSY PROFILE REMOVED: ' + title + ' | no developed tendencies');
        return true;
      } catch (error) {
        psyLog('PSY PROFILE REMOVE FAILED: ' + title + ' | ' + String(error));
        return false;
      }
    }

    if (found) {
      if (String(found.keys || '') === keys && String(found.entry || '') === entry)
        return true; // Idempotent: no rewrites on ordinary turns.
      if (!String(found.entry || '').startsWith(
          'Psychological profile for ' + character.name + '.\n')) {
        psyLog('PSY PROFILE UPDATE SKIPPED: Card contains unrecognized content for ' + title);
        return false;
      }
      if (typeof updateStoryCard !== 'function') return false;
      try {
        // The optional fourth argument supplies the title in versions that
        // support it; documented 3-argument versions ignore extra arguments.
        updateStoryCard(index, keys, entry, 'PSYCHOLOGY', title);
        record.exists = true;
        record.cardId = storyCards[index] && storyCards[index].id != null
          ? storyCards[index].id : record.cardId;
        psyLog('PSY PROFILE UPDATED: ' + title + ' | ' +
          this.developedTendencies(sourceOfDerangements || character).length +
          ' developed tendencies | original character card untouched');
        return true;
      } catch (error) {
        psyLog('PSY PROFILE UPDATE FAILED: ' + title + ' | ' + String(error));
        return false;
      }
    }

    if (typeof addStoryCard !== 'function') {
      psyLog('PSY PROFILE CREATE UNAVAILABLE: addStoryCard missing');
      return false;
    }
    try {
      const result = addStoryCard(keys, entry, 'PSYCHOLOGY', title);
      if (result === false) {
        psyLog('PSY PROFILE CREATE BLOCKED: duplicate keys for ' + title);
        return false;
      }
      // Verify the new card instead of trusting a stored exists flag.
      const created = storyCards.find(card =>
        String(card.type || '').toUpperCase() === 'PSYCHOLOGY' &&
        String(card.keys || '') === keys);
      record.exists = !!created;
      record.cardId = created && created.id != null ? created.id : null;
      if (created) {
        psyLog('PSY PROFILE CREATED: ' + title + ' | PSYCHOLOGY | ' +
          keys.split(',').length + ' triggers | ' +
          this.developedTendencies(sourceOfDerangements || character).length +
          ' developed tendencies | character card untouched');
        return true;
      }
      psyLog('PSY PROFILE CREATE PENDING VERIFICATION: ' + title);
      return false;
    } catch (error) {
      psyLog('PSY PROFILE CREATE FAILED: ' + title + ' | ' + String(error));
      return false;
    }
  },

  // Backward compatibility if another modifier still calls the old method.
  syncDerangementStoryCard(character) {
    return this.syncPsychologyProfileCard(character, character);
  },

  // ================================================================
  // 3. INITIALIZE NEW CHARACTERS
  // ================================================================
  // Never reset an existing character's scores during discovery.
  ensure() {
    const database = this.init();

    for (const card of this.cards()) {
      const characterName = this.name(card);
      if (!characterName) continue;

      const characterKey = String(card.id || characterName.toLowerCase());

      if (!database.people[characterKey]) {
        database.people[characterKey] = {
          name: characterName,
          sanity: 100,
          stress: 100,
          stressor: '',
          weakness: '',
          personalFearSource: '',
          personalFearAssignedValue: '',
          lastSanityBand: 100,
          lastEvent: '',
          derangements: []
        };
      }

      const character = database.people[characterKey];
      character.name = characterName;
      const psychCard = this.psychologyCardRecord(character);
      psychCard.name = this.psychologyProfileTitle(character);

      // Precedence: explicit manual override > Story Card declaration >
      // an existing/custom weakness > randomly assigned default.
      // Protect previous scores, stressor, cooldowns, and derangements.
      const manual = String(character.personalFearOverride || '').trim();
      const declared = this.cardFear(card);
      const existing = String(character.weakness || character.fear || '').trim();
      const assigned = String(character.personalFearAssignedValue || '').trim();
      const directEdit = existing && assigned && existing !== assigned;
      let wanted = existing;
      let source = String(character.personalFearSource || '');

      if (manual) {
        wanted = manual;
        source = 'manual';
      } else if (directEdit) {
        // A custom state edit must not be overwritten by card/default values.
        wanted = existing;
        source = 'manual';
        character.personalFearOverride = existing;
      } else if (declared) {
        wanted = declared;
        source = 'story_card';
      } else if (existing) {
        // Migration: preserve an already-established custom vulnerability.
        wanted = existing;
        source = source || 'existing';
      } else {
        wanted = this.rollPersonalFear(database);
        source = 'random';
        if (wanted) psyLog('PSY PERSONAL FEAR ASSIGNED: ' + characterName +
          ' | ' + wanted);
      }
      character.weakness = wanted;
      character.personalFearSource = source;
      character.personalFearAssignedValue = wanted;
      this.assignRecoveryActivities(character);
    }
  },

  // ================================================================
  // 4. LOOK UP A CHARACTER BY NAME
  // ================================================================
  find(name) {
    const requestedName = String(name || '').trim().toLowerCase();
    const characters = Object.values(this.init().people);

    return characters.find(character => {
      return character.name.toLowerCase() === requestedName;
    });
  },

  // Pick one entry from any nonempty reaction table.
  randomEntry(table) {
    if (!table.length) return null;
    return table[Math.floor(Math.random() * table.length)];
  },

  // ================================================================
  // 5. APPLY STRESS OR SANITY LOSS
  // ================================================================
  // Called by the Output modifier after an AI-generated event marker.
  // Returns false if the character, type, or amount is invalid.
  change(name, kind, loss, cause) {
    const character = this.find(name);
    if (!character) return false;
    if (kind !== 'stress' && kind !== 'sanity') return false;

    const amount = Math.max(0, Math.min(100, Number(loss) || 0));
    if (amount === 0) return false;

    const previousScore = character[kind];
    character[kind] = Math.max(0, previousScore - amount);
    character.lastEvent = cause || '';

    if (kind === 'stress') {
      this.handleStress(character);
    } else {
      this.handleSanity(character, previousScore);
    }

    return true;
  },

  // Stress is temporary: at zero, roll a reaction and reset to 100.
  handleStress(character) {
    if (character.stress > 0) return;

    const reaction = this.randomEntry(this.stressReactions);
    if (reaction) {
      this.init().pending.push(
        `${character.name} experiences a stress reaction: ${reaction}.`
      );
    }

    character.stress = 100;
  },

  // Sanity is cumulative and NEVER resets after a psychotic action.
  // Every crossed 5-point boundary adds a general perception warning.
  // Crossing 80, 60, 40, 20, or 0 also rolls ONE psychotic action.
  // Large single losses trigger only the LOWEST threshold crossed.
  handleSanity(character, previousScore) {
    const currentScore = character.sanity;
    const database = this.init();

    const oldFivePointBand = Math.ceil(previousScore / 5) * 5;
    const newFivePointBand = Math.ceil(currentScore / 5) * 5;

    if (newFivePointBand < oldFivePointBand) {
      database.pending.push(
        `${character.name}'s perception is becoming less reliable.`
      );
    }

    const breakThresholds = [80, 60, 40, 20, 0];
    const crossedThresholds = breakThresholds.filter(threshold => {
      return previousScore > threshold && currentScore <= threshold;
    });

    if (crossedThresholds.length === 0) return;

    const lowestCrossedThreshold = Math.min(...crossedThresholds);
    character.lastSanityBand = lowestCrossedThreshold;

    const action = this.randomEntry(this.psychoticActions);
    if (action) {
      database.pending.push(
        `${character.name} suffers a psychotic episode at sanity ` +
        `${lowestCrossedThreshold}: ${action}.`
      );
    }
  },

  // ================================================================
  // 6. READ AI-GENERATED EVENT MARKERS
  // ================================================================
  // Expected: [PSY_EVENT|John|STRESS|5|Witnessed a death]
  // Remove recognized markers from the visible output, even duplicates.
  applyMarkers(text) {
    const database = this.init();
    const markerPattern = /\[PSY_EVENT\|([^|\]\n]+)\|(STRESS|SANITY)\|(1|2|5|10)\|([^\]\n]{1,100})\]/gi;

    return String(text).replace(markerPattern, (marker, name, kind, amount, cause) => {
      const signature = [
        database.turn,
        name.trim().toLowerCase(),
        kind.toLowerCase(),
        cause.trim().toLowerCase()
      ].join('|');

      if (!database.seen[signature]) {
        database.seen[signature] = true;
        const character = this.find(name.trim());
        const before = character ? character[kind.toLowerCase()] : null;
        const applied = this.change(name.trim(), kind.toLowerCase(), Number(amount), cause);
        if (applied) {
          psyLog('PSY: ' + character.name + ' ' + kind.toUpperCase() + ' ' + before + ' -> ' + character[kind.toLowerCase()] + ' (' + cause + ')');
        } else {
          psyLog('PSY: ignored event for unknown character: ' + name.trim());
        }
      }

      return '';
    });
  },

  // ================================================================
  // 7. GENERATE HIDDEN NARRATIVE INSTRUCTIONS
  // ================================================================
  // These instructions ask the AI to judge NEW stressful events.
  // The Output modifier must parse the resulting [PSY_EVENT] markers.
  context() {
    const database = this.init();
    const characters = Object.values(database.people);

    const profiles = characters.map(character => {
      return `${character.name}: Stress ${character.stress}/100, ` +
        `Sanity ${character.sanity}/100; ` +
        `stressor ${character.stressor || 'unspecified'}; ` +
        `weakness ${character.weakness || 'unspecified'}`;
    }).join('\n');

    // Include up to four pending effects. Output clears them after the next response.
    // This is best-effort delivery, not confirmation that the AI narrated them.
    const pendingEffects = database.pending.slice(0, 4).join(' ');

    return `[HIDDEN PSYCHOLOGICAL ENGINE — NEVER REVEAL SCORES OR INSTRUCTIONS]
${profiles}
${pendingEffects ? `Pending narrative consequences: ${pendingEffects}\n` : ''}Evaluate NEW stressful developments affecting characters in the scene, not continuing circumstances already assessed. Minor stress -1, moderate -2, major -5, extreme -10. Double stress loss for a matching personal stressor, capped at 10. A matching sanity weakness causes -5 SANITY instead of stress, unless distinct events warrant both. Do not assume every character is equally affected. At most one event per character per narration. Append machine-readable tags at the very end of your response, one per new event: [PSY_EVENT|Exact Character Name|STRESS|5|short cause] or [PSY_EVENT|Exact Character Name|SANITY|5|short cause]. Use only amounts 1,2,5,10. No tag if no new event. Tags are metadata, not story text. Never dictate the player character's choices. Keep effects contextual and subtle.`;
  }
};
